BEP-040: Autenticação e Autorização

Implementar sistemas de autenticação de usuários

Login e Logout

Implementando autenticação de usuários.

View de Login

# core/views.py
from django.shortcuts import render, redirect
from django.contrib.auth import authenticate, login
from django.contrib import messages

def login_view(request):
    if request.method == 'POST':
        username = request.POST.get('username')
        password = request.POST.get('password')
        
        # autentica o usuário
        user = authenticate(request, username=username, password=password)
        
        if user is not None:
            login(request, user)  # faz login do usuário
            return redirect('core:home')
        else:
            messages.error(request, 'Usuário ou senha inválidos.')
    
    return render(request, 'login.html')

authenticate(): Verifica credenciais e retorna o objeto User se válido.

login(): Cria uma sessão para o usuário autenticado.

request.user: Após login, contém o usuário autenticado.

View de Logout

# core/views.py
from django.contrib.auth import logout
from django.shortcuts import redirect

def logout_view(request):
    logout(request)  # encerra a sessão do usuário
    return redirect('core:login')

logout(): Remove a sessão do usuário autenticado.

Após logout, request.user será um usuário anônimo.

Verificar Usuário Autenticado

# core/views.py
def minha_view(request):
    if request.user.is_authenticated:
        # usuário está logado
        nome = request.user.username
        return render(request, 'home.html', {'nome': nome})
    else:
        # usuário não está logado
        return redirect('core:login')

request.user.is_authenticated: Retorna True se o usuário está logado.

request.user: Objeto User do usuário atual (ou AnonymousUser se não logado).

URLs

# core/urls.py
from django.urls import path
from .views import login_view, logout_view

app_name = 'core'
urlpatterns = [
    path('login/', login_view, name='login'),
    path('logout/', logout_view, name='logout'),
]
