BEP-021: Composição e Associação entre Objetos

Criando relações entre objetos para modelar sistemas complexos

Exemplo Completo: Sistema de Biblioteca

Aplicando associação e composição juntas

Cenário

Vamos modelar um sistema de biblioteca com:

  • Livro: Tem páginas (Composição) e pode ser emprestado a leitores (Associação)
  • Página: É parte de um livro (Composição)
  • Leitor: Pode pegar livros emprestados (Associação)
  • Biblioteca: Contém livros (Associação)

Implementação Completa

# ===== COMPOSIÇÃO: Página é parte de Livro =====
class Pagina:
    def __init__(self, numero, conteudo):
        self.numero = numero
        self.conteudo = conteudo
    
    def __str__(self):
        return f"Página {self.numero}: {self.conteudo[:50]}..."

class Livro:
    def __init__(self, titulo, autor, paginas_conteudo):
        self.titulo = titulo
        self.autor = autor
        # COMPOSIÇÃO: Páginas são criadas DENTRO do Livro
        self.paginas = []
        for i, conteudo in enumerate(paginas_conteudo, 1):
            self.paginas.append(Pagina(i, conteudo))
        # Páginas não existem sem o livro!
        self.emprestado = False
        self.leitor = None  # ASSOCIAÇÃO: referência ao Leitor
    
    def emprestar(self, leitor):
        """ASSOCIAÇÃO: Cria relação com Leitor"""
        if self.emprestado:
            print(f"Livro já está emprestado para {self.leitor.nome}")
        else:
            self.emprestado = True
            self.leitor = leitor  # Associação criada!
            leitor.livros_emprestados.append(self)  # Associação bidirecional
            print(f"'{self.titulo}' emprestado para {leitor.nome}")
    
    def devolver(self):
        """Remove associação com Leitor"""
        if self.emprestado:
            leitor_anterior = self.leitor.nome
            self.leitor.livros_emprestados.remove(self)
            self.leitor = None
            self.emprestado = False
            print(f"'{self.titulo}' devolvido por {leitor_anterior}")
        else:
            print("Livro não está emprestado")
    
    def info(self):
        print(f"Livro: {self.titulo} - {self.autor}")
        print(f"Páginas: {len(self.paginas)}")
        if self.emprestado:
            print(f"Emprestado para: {self.leitor.nome}")
        else:
            print("Disponível")

# ===== ASSOCIAÇÃO: Leitor usa Livro =====
class Leitor:
    def __init__(self, nome, cpf):
        self.nome = nome
        self.cpf = cpf
        self.livros_emprestados = []  # Associação: lista de Livros
    
    def pegar_emprestado(self, livro):
        """Cria associação com Livro"""
        livro.emprestar(self)
    
    def devolver_livro(self, livro):
        """Remove associação com Livro"""
        livro.devolver()
    
    def listar_livros(self):
        if self.livros_emprestados:
            print(f"\n{self.nome} tem os seguintes livros:")
            for livro in self.livros_emprestados:
                print(f"  - {livro.titulo}")
        else:
            print(f"{self.nome} não tem livros emprestados")

# ===== ASSOCIAÇÃO: Biblioteca contém Livros =====
class Biblioteca:
    def __init__(self, nome):
        self.nome = nome
        self.livros = []  # Associação: lista de Livros
    
    def adicionar_livro(self, livro):
        """Cria associação com Livro"""
        self.livros.append(livro)
        print(f"Livro '{livro.titulo}' adicionado à biblioteca")
    
    def listar_livros(self):
        print(f"\nLivros na {self.nome}:")
        for livro in self.livros:
            status = "Emprestado" if livro.emprestado else "Disponível"
            print(f"  - {livro.titulo} ({status})")

# ===== USO =====

# Criar livro (páginas são criadas automaticamente - COMPOSIÇÃO)
livro1 = Livro("Python para Iniciantes", "João Silva", 
               ["Introdução ao Python", "Variáveis e Tipos", "Estruturas de Controle"])
livro2 = Livro("POO em Python", "Maria Santos",
               ["Classes e Objetos", "Herança", "Polimorfismo"])

# Criar leitores (objetos independentes)
leitor1 = Leitor("Pedro", "123.456.789-00")
leitor2 = Leitor("Ana", "987.654.321-00")

# Criar biblioteca
biblioteca = Biblioteca("Biblioteca Central")

# Adicionar livros à biblioteca (ASSOCIAÇÃO)
biblioteca.adicionar_livro(livro1)
biblioteca.adicionar_livro(livro2)

# Leitor pega livro emprestado (ASSOCIAÇÃO)
leitor1.pegar_emprestado(livro1)
# 'Python para Iniciantes' emprestado para Pedro

leitor2.pegar_emprestado(livro2)
# 'POO em Python' emprestado para Ana

# Listar informações
biblioteca.listar_livros()
leitor1.listar_livros()
leitor2.listar_livros()

# Devolver livro (remove associação)
leitor1.devolver_livro(livro1)
# 'Python para Iniciantes' devolvido por Pedro

Relações Identificadas

Composição:

  • Livro → Páginas: Páginas são criadas dentro do livro e não existem sem ele

Associação:

  • Leitor ↔ Livro: Leitor pode pegar livros emprestados (relação flexível)
  • Biblioteca ↔ Livro: Biblioteca contém livros (relação flexível)
