Padrão Decorator - Implementação Prática

Implementando o padrão Decorator em Python

Antes do padrão: explosão de classes

Vamos usar o exemplo de bebidas e adicionais, mas sem Decorator.

# ❌ Sem Decorator - explosão de classes
class Cafe:
    def custo(self):
        return 5.0

class CafeComLeite(Cafe):
    def custo(self):
        return super().custo() + 2.0

class CafeComAcucar(Cafe):
    def custo(self):
        return super().custo() + 1.0

class CafeComLeiteEAcucar(Cafe):
    def custo(self):
        return super().custo() + 2.0 + 1.0

# E se quiser café com leite, açúcar E chantilly?
# Precisa criar outra classe!
# E se quiser outras combinações? Mais classes!
Problema: Para cada combinação de adicionais, precisa criar uma nova classe. Isso leva a uma explosão de classes e código duplicado.

Com Decorator: composição dinâmica

# ✅ Componente base
class Bebida:
    def __init__(self):
        self._descricao = "Bebida"
    
    def descricao(self):
        return self._descricao
    
    def custo(self):
        return 0.0

# Componente concreto
class Cafe(Bebida):
    def __init__(self):
        super().__init__()
        self._descricao = "Café"
    
    def custo(self):
        return 5.0

# Decorator abstrato
class Adicional(Bebida):
    def __init__(self, bebida):
        self._bebida = bebida
    
    def descricao(self):
        return self._bebida.descricao()
    
    def custo(self):
        return self._bebida.custo()

# Decorators concretos
class Leite(Adicional):
    def descricao(self):
        return self._bebida.descricao() + ", Leite"
    
    def custo(self):
        return self._bebida.custo() + 2.0

class Acucar(Adicional):
    def descricao(self):
        return self._bebida.descricao() + ", Açúcar"
    
    def custo(self):
        return self._bebida.custo() + 1.0

class Chantilly(Adicional):
    def descricao(self):
        return self._bebida.descricao() + ", Chantilly"
    
    def custo(self):
        return self._bebida.custo() + 3.0

# Uso - composição dinâmica!
cafe = Cafe()
print(f"{cafe.descricao()}: R$ {cafe.custo():.2f}")
# Café: R$ 5.00

cafe_com_leite = Leite(Cafe())
print(f"{cafe_com_leite.descricao()}: R$ {cafe_com_leite.custo():.2f}")
# Café, Leite: R$ 7.00

cafe_completo = Chantilly(Acucar(Leite(Cafe())))
print(f"{cafe_completo.descricao()}: R$ {cafe_completo.custo():.2f}")
# Café, Leite, Açúcar, Chantilly: R$ 11.00
✅ Vantagem: Pode combinar qualquer quantidade de adicionais dinamicamente, sem precisar criar novas classes para cada combinação!

Exemplo: Streams de Dados

# Componente base
class Stream:
    def ler(self):
        pass
    
    def escrever(self, dados):
        pass

# Componente concreto
class FileStream(Stream):
    def __init__(self, arquivo):
        self.arquivo = arquivo
    
    def ler(self):
        return f"Dados de {self.arquivo}"
    
    def escrever(self, dados):
        print(f"Escrevendo em {self.arquivo}: {dados}")

# Decorator
class StreamDecorator(Stream):
    def __init__(self, stream):
        self._stream = stream
    
    def ler(self):
        return self._stream.ler()
    
    def escrever(self, dados):
        self._stream.escrever(dados)

# Decorators concretos
class CompressaoStream(StreamDecorator):
    def escrever(self, dados):
        dados_comprimidos = f"[COMPRIMIDO] {dados}"
        self._stream.escrever(dados_comprimidos)
    
    def ler(self):
        dados = self._stream.ler()
        return dados.replace("[COMPRIMIDO] ", "")

class CriptografiaStream(StreamDecorator):
    def escrever(self, dados):
        dados_criptografados = f"[CRIPTOGRAFADO] {dados}"
        self._stream.escrever(dados_criptografados)
    
    def ler(self):
        dados = self._stream.ler()
        return dados.replace("[CRIPTOGRAFADO] ", "")

# Uso - combinando decorators
stream_base = FileStream("dados.txt")

# Apenas compressão
stream_comp = CompressaoStream(stream_base)
stream_comp.escrever("Dados importantes")

# Compressão + Criptografia
stream_seguro = CriptografiaStream(CompressaoStream(FileStream("seguro.txt")))
stream_seguro.escrever("Dados secretos")

Exemplo: Formatação de Texto

# Componente base
class Texto:
    def __init__(self, conteudo):
        self._conteudo = conteudo
    
    def renderizar(self):
        return self._conteudo

# Decorator
class Formatador(Texto):
    def __init__(self, texto):
        self._texto = texto
    
    def renderizar(self):
        return self._texto.renderizar()

# Decorators concretos
class Negrito(Formatador):
    def renderizar(self):
        return f"{self._texto.renderizar()}"

class Italico(Formatador):
    def renderizar(self):
        return f"{self._texto.renderizar()}"

class Sublinhado(Formatador):
    def renderizar(self):
        return f"{self._texto.renderizar()}"

# Uso - combinando formatações
texto = Texto("Olá Mundo")

texto_negrito = Negrito(texto)
print(texto_negrito.renderizar())
# Olá Mundo

texto_completo = Sublinhado(Italico(Negrito(Texto("Python"))))
print(texto_completo.renderizar())

Vantagens do Decorator

  • ✅ Flexibilidade: Adiciona funcionalidades dinamicamente
  • ✅ Composição: Combina múltiplas funcionalidades facilmente
  • ✅ Extensibilidade: Fácil adicionar novos decorators
  • ✅ Separação: Cada decorator tem responsabilidade única
  • ✅ Reutilização: Decorators podem ser reutilizados

Dicas de Implementação

💡 Lembre-se:
  • Decorator mantém referência ao objeto decorado
  • Decorator delega chamadas ao objeto decorado
  • Múltiplos decorators podem ser encadeados
  • Ordem dos decorators importa (mais externo é aplicado primeiro)
