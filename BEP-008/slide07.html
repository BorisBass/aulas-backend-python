Organize seu código em blocos reutilizáveis

Retorno de Valores

🔄 Comando return

O comando return envia um valor de volta para quem chamou a função.

# Função sem return (retorna None) def cumprimentar(): print("Olá, mundo!") # Função com return def obter_saudacao(): return "Olá, mundo!" # Testando as funções cumprimentar() # Imprime: Olá, mundo! mensagem = obter_saudacao() # Retorna: "Olá, mundo!" print(mensagem)

📊 Tipos de Retorno

# Retornando números def calcular_quadrado(numero): return numero * numero # Retornando texto def obter_nome_completo(nome, sobrenome): return f"{nome} {sobrenome}" # Retornando booleanos def verificar_par(numero): return numero % 2 == 0 # Retornando listas def criar_lista_pares(): return [2, 4, 6, 8, 10] # Testando as funções quadrado = calcular_quadrado(5) nome = obter_nome_completo("João", "Silva") par = verificar_par(8) pares = criar_lista_pares() print(f"Quadrado: {quadrado}") print(f"Nome: {nome}") print(f"É par: {par}") print(f"Pares: {pares}")

🎯 Múltiplos Returns

# Função com múltiplos returns def verificar_idade(idade): if idade < 0: return "Idade inválida" elif idade < 13: return "Criança" elif idade < 20: return "Adolescente" elif idade < 60: return "Adulto" else: return "Idoso" # Testando diferentes idades print(verificar_idade(5)) # Criança print(verificar_idade(15)) # Adolescente print(verificar_idade(25)) # Adulto print(verificar_idade(65)) # Idoso print(verificar_idade(-1)) # Idade inválida

📦 Retornando Múltiplos Valores

# Função que retorna múltiplos valores def calcular_estatisticas(numeros): soma = sum(numeros) media = soma / len(numeros) maior = max(numeros) menor = min(numeros) return soma, media, maior, menor # Usando a função lista = [1, 2, 3, 4, 5] soma, media, maior, menor = calcular_estatisticas(lista) print(f"Lista: {lista}") print(f"Soma: {soma}") print(f"Média: {media:.2f}") print(f"Maior: {maior}") print(f"Menor: {menor}")

⚠️ Funções sem Return

# Função sem return (retorna None) def mostrar_informacoes(): print("Nome: João Silva") print("Idade: 25 anos") print("Cidade: Salvador") # Função com return None (explícito) def processar_dados(): print("Processando dados...") return None # Testando as funções resultado1 = mostrar_informacoes() resultado2 = processar_dados() print(f"Resultado 1: {resultado1}") # None print(f"Resultado 2: {resultado2}") # None

🔄 Usando Valores Retornados

# Função que calcula e retorna def calcular_media(nota1, nota2, nota3): return (nota1 + nota2 + nota3) / 3 # Usando o valor retornado media = calcular_media(8.5, 7.0, 9.2) print(f"Média: {media:.2f}") # Usando em cálculos nova_media = media + 1.0 print(f"Nova média: {nova_media:.2f}") # Usando em condições if media >= 7.0: print("Aprovado!") else: print("Reprovado!")

Dicas Importantes

  • Return único: Uma função pode ter múltiplos returns
  • Valor padrão: Funções sem return retornam None
  • Múltiplos valores: Retornados como tupla
  • Uso imediato: Pode usar o valor retornado diretamente
💡 Exemplo Prático
# Função para verificar aprovação def verificar_aprovacao(nota1, nota2, nota3): media = (nota1 + nota2 + nota3) / 3 if media >= 7.0: return True, media else: return False, media # Testando a função aprovado, media = verificar_aprovacao(8.5, 7.0, 9.2) if aprovado: print(f"Aprovado com média {media:.2f}") else: print(f"Reprovado com média {media:.2f}")
