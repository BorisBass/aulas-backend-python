🎯 Exercício 1: Lista de Compras

Crie um sistema de lista de compras interativo.

# Lista de compras compras = [] print("=== LISTA DE COMPRAS ===") while True: print(" Opções:") print("1. Adicionar item") print("2. Remover item") print("3. Listar itens") print("4. Sair") opcao = input("Escolha uma opção: ") if opcao == "1": item = input("Digite o item: ") compras.append(item) print(f"'{item}' adicionado à lista!") elif opcao == "2": if compras: print("Itens na lista:") for i, item in enumerate(compras): print(f"{i}: {item}") try: indice = int(input("Índice do item para remover: ")) item_removido = compras.pop(indice) print(f"'{item_removido}' removido da lista!") except (ValueError, IndexError): print("Índice inválido!") else: print("Lista vazia!") elif opcao == "3": if compras: print(" Itens na lista:") for i, item in enumerate(compras): print(f"{i}: {item}") else: print("Lista vazia!") elif opcao == "4": print("Saindo...") break else: print("Opção inválida!")

🎯 Exercício 2: Análise de Notas

Analise uma lista de notas e calcule estatísticas.

# Análise de notas notas = [] print("Digite as notas (digite -1 para parar):") while True: try: nota = float(input("Nota: ")) if nota == -1: break if 0 <= nota <= 10: notas.append(nota) else: print("Nota deve estar entre 0 e 10!") except ValueError: print("Digite um número válido!") if notas: print(f" === ANÁLISE DE NOTAS ===") print(f"Notas: {notas}") print(f"Quantidade: {len(notas)}") print(f"Maior nota: {max(notas)}") print(f"Menor nota: {min(notas)}") print(f"Média: {sum(notas)/len(notas):.2f}") # Contar aprovações aprovados = sum(1 for nota in notas if nota >= 7) print(f"Aprovados: {aprovados} ({aprovados/len(notas)*100:.1f}%)") else: print("Nenhuma nota válida foi digitada!")

🎯 Exercício 3: Busca em Lista

Implemente um sistema de busca em lista.

# Sistema de busca nomes = ["Ana", "Bruno", "Carlos", "Diana", "Eduardo"] print("Lista de nomes:", nomes) nome_busca = input("Digite um nome para buscar: ") if nome_busca in nomes: indice = nomes.index(nome_busca) print(f"'{nome_busca}' encontrado no índice {indice}") else: print(f"'{nome_busca}' não encontrado na lista!") # Sugerir nomes similares sugestoes = [nome for nome in nomes if nome_busca.lower() in nome.lower()] if sugestoes: print(f"Nomes similares: {sugestoes}")
