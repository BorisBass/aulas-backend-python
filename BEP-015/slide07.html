Exercício 1: INNER JOIN

Objetivo: Mostrar livros com informações do autor e categoria

Resultado esperado: Título, Autor, Categoria, Ano, Preço

-- Sua consulta aqui
SELECT 
    l.titulo,
    a.nome as autor,
    c.nome as categoria,
    l.ano,
    l.preco
FROM livros l
INNER JOIN autores a ON l.autor_id = a.id
INNER JOIN categorias c ON l.categoria_id = c.id;

Exercício 2: LEFT JOIN

Objetivo: Mostrar TODOS os autores, mesmo os sem livros

Resultado esperado: Nome do autor, Nacionalidade, Título (NULL se não tiver)

-- Sua consulta aqui
SELECT 
    a.nome as autor,
    a.nacionalidade,
    l.titulo
FROM autores a
LEFT JOIN livros l ON a.id = l.autor_id;

Exercício 3: Subconsulta

Objetivo: Encontrar livros com preço acima da média

Resultado esperado: Título, Preço, Autor

-- Sua consulta aqui
SELECT 
    l.titulo,
    l.preco,
    a.nome as autor
FROM livros l
INNER JOIN autores a ON l.autor_id = a.id
WHERE l.preco > (
    SELECT AVG(preco) 
    FROM livros
);

Exercício 4: GROUP BY

Objetivo: Calcular estatísticas por categoria

Resultado esperado: Categoria, Total de livros, Preço médio, Preço total

-- Sua consulta aqui
SELECT 
    c.nome as categoria,
    COUNT(*) as total_livros,
    AVG(l.preco) as preco_medio,
    SUM(l.preco) as preco_total
FROM livros l
INNER JOIN categorias c ON l.categoria_id = c.id
GROUP BY c.nome;

Exercício 5: ORDER BY

Objetivo: Mostrar livros ordenados por preço (maior para menor)

Resultado esperado: Título, Preço, Autor (ordenado por preço)

-- Sua consulta aqui
SELECT 
    l.titulo,
    l.preco,
    a.nome as autor
FROM livros l
INNER JOIN autores a ON l.autor_id = a.id
ORDER BY l.preco DESC;

Exercício 6: Combinado

Objetivo: Mostrar autores com mais livros, ordenados por quantidade

Resultado esperado: Autor, Quantidade de livros, Preço médio

-- Sua consulta aqui
SELECT 
    a.nome as autor,
    COUNT(*) as quantidade_livros,
    AVG(l.preco) as preco_medio
FROM autores a
INNER JOIN livros l ON a.id = l.autor_id
GROUP BY a.nome
ORDER BY quantidade_livros DESC;
