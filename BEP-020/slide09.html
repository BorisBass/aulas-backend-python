BEP-020: Herança e Polimorfismo

Reutilização e extensão de código através de herança e polimorfismo

Exemplo: Veiculos e Subclasses

Sistema completo com herança e polimorfismo

Sistema Completo de Veículos

class Veiculo:
    def __init__(self, marca, modelo, ano):
        self.marca = marca
        self.modelo = modelo
        self.ano = ano
        self.velocidade = 0
        self.ligado = False
    
    def ligar(self):
        self.ligado = True
        return f"{self.modelo} ligado"
    
    def desligar(self):
        self.ligado = False
        self.velocidade = 0
        return f"{self.modelo} desligado"
    
    def acelerar(self, incremento=10):
        if self.ligado:
            self.velocidade += incremento
            return f"{self.modelo} acelerou para {self.velocidade} km/h"
        return f"{self.modelo} precisa estar ligado"
    
    def frear(self):
        self.velocidade = max(0, self.velocidade - 10)
        return f"{self.modelo} freou para {self.velocidade} km/h"
    
    def informacoes(self):
        return f"{self.marca} {self.modelo} ({self.ano}) - {self.velocidade} km/h"

class Carro(Veiculo):
    def __init__(self, marca, modelo, ano, portas):
        super().__init__(marca, modelo, ano)
        self.portas = portas
    
    def acelerar(self, incremento=15):  # Polimorfismo
        if self.ligado:
            self.velocidade += incremento
            return f"{self.modelo} acelerou suavemente para {self.velocidade} km/h"
        return f"{self.modelo} precisa estar ligado"
    
    def abrir_porta(self, numero):
        return f"Porta {numero} do {self.modelo} aberta"

class Moto(Veiculo):
    def __init__(self, marca, modelo, ano, cilindradas):
        super().__init__(marca, modelo, ano)
        self.cilindradas = cilindradas
    
    def acelerar(self, incremento=25):  # Polimorfismo
        if self.ligado:
            self.velocidade += incremento
            return f"{self.modelo} acelerou rapidamente para {self.velocidade} km/h"
        return f"{self.modelo} precisa estar ligado"
    
    def empinar(self):
        if self.velocidade > 30:
            return f"{self.modelo} empinando!"
        return "Velocidade insuficiente para empinar"

class Caminhao(Veiculo):
    def __init__(self, marca, modelo, ano, capacidade_carga):
        super().__init__(marca, modelo, ano)
        self.capacidade_carga = capacidade_carga
        self.carga_atual = 0
    
    def acelerar(self, incremento=5):  # Polimorfismo: acelera mais devagar
        if self.ligado:
            self.velocidade += incremento
            return f"{self.modelo} acelerou lentamente para {self.velocidade} km/h"
        return f"{self.modelo} precisa estar ligado"
    
    def carregar(self, peso):
        if self.carga_atual + peso <= self.capacidade_carga:
            self.carga_atual += peso
            return f"{self.modelo} carregou {peso} kg"
        return "Capacidade de carga excedida"

Demonstração de Polimorfismo

# Criando diferentes veículos
veiculos = [
    Carro("Toyota", "Corolla", 2023, 4),
    Moto("Honda", "CB600", 2022, 600),
    Caminhao("Volvo", "FH", 2021, 10000)
]

# Ligando todos
for veiculo in veiculos:
    print(veiculo.ligar())

# Acelerando todos (comportamento diferente!)
for veiculo in veiculos:
    print(veiculo.acelerar())
    print(veiculo.acelerar())
    print()

# Saída mostra que cada tipo acelera de forma diferente:
# Corolla acelerou suavemente para 15 km/h
# CB600 acelerou rapidamente para 25 km/h
# FH acelerou lentamente para 5 km/h

Análise do Exemplo

  • Herança: Todas as subclasses herdam atributos e métodos de Veiculo
  • Polimorfismo: O método acelerar() se comporta diferente em cada tipo
  • Extensão: Cada subclasse adiciona métodos específicos (abrir_porta, empinar, carregar)
  • Reutilização: Código comum (ligar, desligar, frear) definido uma vez
