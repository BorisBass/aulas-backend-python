Exemplo Prático: Loja Online

Vamos criar uma estrutura simples para uma loja online com clientes e produtos.

-- 1. Criar tabela de clientes
CREATE TABLE clientes (
    id INTEGER PRIMARY KEY,
    nome TEXT NOT NULL,
    email TEXT,
    cidade TEXT
);

-- 2. Criar tabela de produtos
CREATE TABLE produtos (
    id INTEGER PRIMARY KEY,
    nome TEXT NOT NULL,
    preco REAL,
    estoque INTEGER
);

Inserir Dados de Exemplo

-- Inserir clientes
INSERT INTO clientes (nome, email, cidade) VALUES
('João Silva', 'joao@email.com', 'Salvador'),
('Maria Santos', 'maria@email.com', 'Feira de Santana'),
('Pedro Costa', 'pedro@email.com', 'Salvador');

-- Inserir produtos
INSERT INTO produtos (nome, preco, estoque) VALUES
('Notebook Dell', 2500.00, 15),
('Mouse Logitech', 89.90, 50),
('Teclado Mecânico', 199.90, 25);

Consultas Práticas

-- Ver todos os clientes
SELECT * FROM clientes;

-- Ver produtos com preço menor que 200
SELECT nome, preco FROM produtos WHERE preco < 200;

-- Ver clientes de Salvador
SELECT nome, email FROM clientes WHERE cidade = 'Salvador';

-- Ver produtos ordenados por preço
SELECT nome, preco FROM produtos ORDER BY preco DESC;

  • Sequência: CREATE → INSERT → SELECT
