Python (sqlite3)

import sqlite3
with sqlite3.connect("aula.db") as conn:
    conn.execute("PRAGMA foreign_keys = ON;")
    conn.execute("INSERT INTO clientes (nome, email) VALUES (?, ?)", ("Nova Pessoa", None))
