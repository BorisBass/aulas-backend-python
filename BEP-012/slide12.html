Sistema de Cadastro em Python

O que vamos criar?

Sistema de Cadastro com funcionalidades de Create (Criar) e Read (Ler) usando os comandos SQL que já aprendemos.

Prática: Vamos criar um sistema de cadastro de clientes com menu interativo usando INSERT e SELECT.

1. Configuração e Funções Auxiliares

Conceito: Primeiro vamos configurar o banco e criar funções auxiliares para o sistema.

Prática: Configuração inicial e funções para conectar ao banco e exibir menus.

Célula 1: Configuração

import sqlite3
import os

def conectar_banco():
    """Conecta ao banco de dados"""
    try:
        conn = sqlite3.connect('clientes.db')
        cursor = conn.cursor()
        
        # Criar tabela se não existir
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS clientes (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                nome TEXT NOT NULL,
                email TEXT UNIQUE,
                telefone TEXT,
                cidade TEXT,
                data_cadastro DATE DEFAULT CURRENT_DATE
            )
        ''')
        conn.commit()
        print("✅ Banco de dados conectado e tabela criada!")
        return conn, cursor
    except sqlite3.Error as e:
        print(f"❌ Erro ao conectar: {e}")
        return None, None

def exibir_menu():
    """Exibe o menu principal"""
    print("\n" + "="*50)
    print("🏪 SISTEMA DE CADASTRO DE CLIENTES")
    print("="*50)
    print("1. 📝 Cadastrar novo cliente")
    print("2. 📋 Listar todos os clientes")
    print("3. 🔍 Buscar cliente por nome")
    print("4. 📊 Estatísticas")
    print("0. 🚪 Sair")
    print("="*50)

2. CREATE - Cadastrar Cliente

Conceito: Função para inserir novos clientes no banco de dados.

Prática: Validação de dados e inserção com tratamento de erros.

Célula 2: Cadastrar Cliente

def cadastrar_cliente(cursor, conn):
    """Cadastra um novo cliente"""
    try:
        print("\n📝 CADASTRO DE NOVO CLIENTE")
        print("-" * 30)
        
        nome = input("Nome completo: ").strip()
        if not nome:
            print("❌ Nome é obrigatório!")
            return
        
        email = input("Email: ").strip()
        telefone = input("Telefone: ").strip()
        cidade = input("Cidade: ").strip()
        
        # Inserir no banco
        cursor.execute('''
            INSERT INTO clientes (nome, email, telefone, cidade)
            VALUES (?, ?, ?, ?)
        ''', (nome, email, telefone, cidade))
        
        conn.commit()
        print(f"✅ Cliente '{nome}' cadastrado com sucesso!")
        
    except sqlite3.IntegrityError:
        print("❌ Erro: Email já cadastrado!")
    except sqlite3.Error as e:
        print(f"❌ Erro no banco: {e}")
    except Exception as e:
        print(f"❌ Erro inesperado: {e}")

3. READ - Listar Clientes

Conceito: Função para exibir todos os clientes cadastrados.

Prática: Consulta SELECT com formatação de saída.

Célula 3: Listar Clientes

def listar_clientes(cursor):
    """Lista todos os clientes"""
    try:
        cursor.execute("SELECT * FROM clientes ORDER BY nome")
        clientes = cursor.fetchall()
        
        if not clientes:
            print("📭 Nenhum cliente cadastrado!")
            return
        
        print(f"\n📋 LISTA DE CLIENTES ({len(clientes)} cadastrados)")
        print("-" * 80)
        print(f"{'ID':<3} {'Nome':<25} {'Email':<25} {'Cidade':<15} {'Data'}")
        print("-" * 80)
        
        for cliente in clientes:
            id_cliente, nome, email, telefone, cidade, data = cliente
            email_display = email if email else "Não informado"
            cidade_display = cidade if cidade else "Não informado"
            print(f"{id_cliente:<3} {nome:<25} {email_display:<25} {cidade_display:<15} {data}")
            
    except sqlite3.Error as e:
        print(f"❌ Erro ao listar clientes: {e}")

4. READ - Buscar Cliente

Conceito: Função para buscar clientes por nome usando LIKE.

Prática: Consulta com filtro e tratamento de resultados vazios.

Célula 4: Buscar Cliente

def buscar_cliente(cursor):
    """Busca cliente por nome"""
    try:
        nome_busca = input("\n🔍 Digite o nome para buscar: ").strip()
        if not nome_busca:
            print("❌ Digite um nome para buscar!")
            return
        
        cursor.execute("SELECT * FROM clientes WHERE nome LIKE ? ORDER BY nome", 
                      (f"%{nome_busca}%",))
        clientes = cursor.fetchall()
        
        if not clientes:
            print(f"📭 Nenhum cliente encontrado com '{nome_busca}'")
            return
        
        print(f"\n🔍 RESULTADO DA BUSCA por '{nome_busca}' ({len(clientes)} encontrados)")
        print("-" * 80)
        print(f"{'ID':<3} {'Nome':<25} {'Email':<25} {'Cidade':<15} {'Data'}")
        print("-" * 80)
        
        for cliente in clientes:
            id_cliente, nome, email, telefone, cidade, data = cliente
            email_display = email if email else "Não informado"
            cidade_display = cidade if cidade else "Não informado"
            print(f"{id_cliente:<3} {nome:<25} {email_display:<25} {cidade_display:<15} {data}")
            
    except sqlite3.Error as e:
        print(f"❌ Erro ao buscar cliente: {e}")

5. Estatísticas

Conceito: Função para mostrar estatísticas do banco de dados.

Prática: Consultas agregadas com COUNT e GROUP BY.

Célula 5: Estatísticas

def mostrar_estatisticas(cursor):
    """Mostra estatísticas do banco"""
    try:
        print("\n📊 ESTATÍSTICAS DO SISTEMA")
        print("-" * 30)
        
        # Total de clientes
        cursor.execute("SELECT COUNT(*) FROM clientes")
        total = cursor.fetchone()[0]
        print(f"👥 Total de clientes: {total}")
        
        # Clientes por cidade
        cursor.execute('''
            SELECT cidade, COUNT(*) as quantidade 
            FROM clientes 
            WHERE cidade IS NOT NULL AND cidade != ''
            GROUP BY cidade 
            ORDER BY quantidade DESC
        ''')
        cidades = cursor.fetchall()
        
        if cidades:
            print(f"\n🏙️ Clientes por cidade:")
            for cidade, qtd in cidades:
                print(f"  {cidade}: {qtd} cliente(s)")
        
        # Clientes cadastrados hoje
        cursor.execute("SELECT COUNT(*) FROM clientes WHERE data_cadastro = DATE('now')")
        hoje = cursor.fetchone()[0]
        print(f"\n📅 Cadastrados hoje: {hoje}")
        
    except sqlite3.Error as e:
        print(f"❌ Erro ao gerar estatísticas: {e}")

6. Programa Principal

Conceito: Loop principal que executa o menu e chama as funções.

Prática: Menu interativo com tratamento de opções inválidas.

Célula 6: Programa Principal

def main():
    """Função principal do programa"""
    print("🚀 Iniciando Sistema de Cadastro de Clientes...")
    
    # Conectar ao banco
    conn, cursor = conectar_banco()
    if not conn:
        print("❌ Não foi possível conectar ao banco. Encerrando...")
        return
    
    try:
        while True:
            exibir_menu()
            opcao = input("👉 Escolha uma opção: ").strip()
            
            if opcao == '1':
                cadastrar_cliente(cursor, conn)
            elif opcao == '2':
                listar_clientes(cursor)
            elif opcao == '3':
                buscar_cliente(cursor)
            elif opcao == '4':
                mostrar_estatisticas(cursor)
            elif opcao == '0':
                print("\n👋 Obrigado por usar o sistema! Até logo!")
                break
            else:
                print("❌ Opção inválida! Tente novamente.")
            
            input("\n⏸️ Pressione Enter para continuar...")
    
    except KeyboardInterrupt:
        print("\n\n👋 Programa interrompido pelo usuário.")
    except Exception as e:
        print(f"\n❌ Erro inesperado: {e}")
    finally:
        # Sempre fechar a conexão
        if conn:
            conn.close()
            print("🔒 Conexão com banco fechada.")

# Executar o programa
if __name__ == "__main__":
    main()
