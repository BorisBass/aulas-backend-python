Exceções genéricas vs customizadas

Versão 1: Exceções Genéricas

# Tratamento genérico
def cadastrar_aluno(cursor, conn):
    try:
        # ... código ...
        cursor.execute('''
            INSERT INTO alunos (nome, idade, curso, nota)
            VALUES (?, ?, ?, ?)
        ''', (nome, idade, curso, nota))
        conn.commit()
    except sqlite3.Error as e:
        print(f"❌ Erro no banco: {e}")
    except Exception as e:
        print(f"❌ Erro inesperado: {e}")

# Problema: mensagens genéricas, difícil identificar tipo de erro

Versão 2: Exceções Customizadas

# Exceções customizadas (BEP-022)
class ErroSistema(Exception):
    """Exceção base"""
    pass

class DadosInvalidosError(ErroSistema):
    """Erro quando dados são inválidos"""
    def __init__(self, campo, valor=None, motivo=None):
        # Mensagem específica e clara
        super().__init__(f"Campo '{campo}': {motivo}")

class AlunoNaoEncontradoError(ErroSistema):
    """Erro quando aluno não existe"""
    pass

# Uso: tratamento específico
try:
    aluno = Aluno(nome="", nota=15.0)  # Dados inválidos
except DadosInvalidosError as e:
    print(f"❌ {e}")  # Mensagem clara e específica

Vantagens das Exceções Customizadas

  • ✅ Mensagens claras: Específicas para cada tipo de erro
  • ✅ Tratamento específico: Diferentes ações para diferentes erros
  • ✅ Hierarquia: Exceções organizadas (BEP-022)
  • ✅ Manutenção: Fácil identificar e corrigir problemas
