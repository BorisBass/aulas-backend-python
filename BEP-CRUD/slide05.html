Versão 1: Validação Manual

# Validação repetida em cada função
def cadastrar_aluno(cursor, conn):

    if not nome:
        print("❌ Nome é obrigatório!")

        try:

        try:

            if nota < 0 or nota > 10:

Problema: Código duplicado, fácil esquecer validações

Versão 2: Validação Automática

# Validação centralizada na classe Aluno (BEP-019)
class Aluno:
    def __init__(self, nome, idade=None, nota=None):
        # Atributos privados (convenção Python)
        self.__nome = ""
        self.nome = nome  # Usa o setter para validar
        
        self.__nota = 0
        if nota is not None:
            self.nota = nota  # Usa o setter para validar
    
    @property
    def nome(self):
        return self.__nome
    
    @nome.setter
    def nome(self, valor):
        if not valor or not valor.strip():
            raise ValueError("Nome é obrigatório")
        self.__nome = valor
    
    @property
    def nota(self):
        return self.__nota
    
    @nota.setter
    def nota(self, valor):
        if valor is not None and (valor < 0 or valor > 10):
            raise ValueError("Nota deve estar entre 0 e 10")
        self.__nota = valor

# Uso: validação automática
def _cadastrar_aluno(self):
    nome = input("Nome: ").strip()
    nota_str = input("Nota: ").strip()
    nota = float(nota_str) if nota_str else None
    
    # Validação acontece automaticamente aqui!
    aluno = Aluno(nome=nome, nota=nota)
    self.repository.criar(aluno)

Vantagem: Validação centralizada com @property/@setter (forma pythônica - BEP-019), impossível esquecer

Comparação

  • ✅ Versão 1: Validação manual, código duplicado, fácil esquecer
  • ✅ Versão 2: Validação automática, centralizada, sempre aplicada
  • ✅ Manutenção: Versão 2 é mais fácil de manter (mudar validação em um lugar só)
