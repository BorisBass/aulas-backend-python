Melhorando a segurança e robustez do código

Try-Except Dentro de Classes

Tratamento de exceções em métodos de classes

🏗️ Estrutura Básica

Em POO, podemos tratar exceções dentro dos métodos das classes:

class Calculadora:
    def dividir(self, a, b):
        try:
            return a / b
        except ZeroDivisionError:
            print("❌ Erro: Divisão por zero!")
            return None
    
    def raiz_quadrada(self, numero):
        try:
            if numero < 0:
                raise ValueError("Número negativo não tem raiz real!")
            return numero ** 0.5
        except ValueError as e:
            print(f"❌ Erro: {e}")
            return None

calc = Calculadora()
print(calc.dividir(10, 0))        # Erro tratado
print(calc.raiz_quadrada(-4))     # Erro tratado

💡 Exemplo: Classe Conta Bancária

class ContaBancaria:
    def __init__(self, titular, saldo_inicial=0):
        self.titular = titular
        self.saldo = saldo_inicial
    
    def depositar(self, valor):
        try:
            if valor <= 0:
                raise ValueError("Valor deve ser positivo!")
            self.saldo += valor
            print(f"✅ Depósito de R$ {valor:.2f} realizado")
            return True
        except ValueError as e:
            print(f"❌ Erro no depósito: {e}")
            return False
    
    def sacar(self, valor):
        try:
            if valor <= 0:
                raise ValueError("Valor deve ser positivo!")
            if valor > self.saldo:
                raise ValueError("Saldo insuficiente!")
            self.saldo -= valor
            print(f"✅ Saque de R$ {valor:.2f} realizado")
            return True
        except ValueError as e:
            print(f"❌ Erro no saque: {e}")
            return False
    
    def transferir(self, destino, valor):
        try:
            if self.sacar(valor):
                destino.depositar(valor)
                print(f"✅ Transferência de R$ {valor:.2f} realizada")
                return True
            return False
        except AttributeError:
            print("❌ Erro: Conta destino inválida!")
            return False

# Uso
conta1 = ContaBancaria("Ana", 1000)
conta2 = ContaBancaria("João", 500)

conta1.sacar(1500)        # Erro: Saldo insuficiente
conta1.depositar(-100)    # Erro: Valor inválido
conta1.transferir(conta2, 200)  # Sucesso!

🔄 Retornando Valores ou Exceções

class Validador:
    def validar_idade(self, idade_str):
        try:
            idade = int(idade_str)
            if idade < 0:
                raise ValueError("Idade não pode ser negativa!")
            if idade > 150:
                raise ValueError("Idade inválida!")
            return idade
        except ValueError as e:
            print(f"❌ Erro: {e}")
            return None
        except Exception as e:
            print(f"❌ Erro inesperado: {e}")
            return None
    
    def validar_email(self, email):
        try:
            if '@' not in email:
                raise ValueError("Email inválido! Deve conter @")
            if '.' not in email.split('@')[1]:
                raise ValueError("Email inválido! Deve conter domínio")
            return True
        except ValueError as e:
            print(f"❌ Erro: {e}")
            return False

validador = Validador()
idade = validador.validar_idade("25")      # ✅ 25
idade = validador.validar_idade("abc")     # ❌ Erro
email_ok = validador.validar_email("ana@email.com")  # ✅ True

Boa Prática

Em métodos de classes, sempre trate exceções e retorne valores apropriados (True/False, None, ou lançe exceções customizadas) para que o código que chama o método saiba como lidar com o resultado.
