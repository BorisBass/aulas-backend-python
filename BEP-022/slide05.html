BEP-022: Tratamento de Exceções em POO

Melhorando a segurança e robustez do código

Criando Exceções Customizadas

Como criar suas próprias exceções

📝 Forma Simples

# Herdar de Exception
class MinhaExcecao(Exception):
    pass

# Usar
raise MinhaExcecao("Mensagem de erro")

💡 Forma Completa (Recomendada)

class SaldoInsuficienteError(Exception):
    """Exceção lançada quando saldo é insuficiente"""
    
    def __init__(self, saldo_atual, valor_solicitado):
        self.saldo_atual = saldo_atual
        self.valor_solicitado = valor_solicitado
        self.mensagem = (
            f"Saldo insuficiente! "
            f"Disponível: R$ {saldo_atual:.2f}, "
            f"Solicitado: R$ {valor_solicitado:.2f}"
        )
        super().__init__(self.mensagem)

# Uso
try:
    raise SaldoInsuficienteError(100, 500)
except SaldoInsuficienteError as e:
    print(f"❌ {e}")
    print(f"   Saldo atual: R$ {e.saldo_atual:.2f}")
    print(f"   Valor solicitado: R$ {e.valor_solicitado:.2f}")

🏗️ Exemplo Completo: Sistema Bancário

# Exceções customizadas
class SaldoInsuficienteError(Exception):
    """Erro quando saldo é insuficiente"""
    def __init__(self, saldo, valor):
        self.saldo = saldo
        self.valor = valor
        super().__init__(f"Saldo insuficiente! Disponível: R$ {saldo:.2f}")

class ValorInvalidoError(Exception):
    """Erro quando valor é inválido"""
    def __init__(self, valor):
        self.valor = valor
        super().__init__(f"Valor inválido: {valor}")

class ContaNaoEncontradaError(Exception):
    """Erro quando conta não é encontrada"""
    pass

# Classe usando as exceções
class ContaBancaria:
    def __init__(self, numero, saldo_inicial=0):
        self.numero = numero
        self.saldo = saldo_inicial
    
    def sacar(self, valor):
        if valor <= 0:
            raise ValorInvalidoError(valor)
        if valor > self.saldo:
            raise SaldoInsuficienteError(self.saldo, valor)
        self.saldo -= valor
        return True
    
    def depositar(self, valor):
        if valor <= 0:
            raise ValorInvalidoError(valor)
        self.saldo += valor
        return True

# Uso
conta = ContaBancaria("001", 1000)

try:
    conta.sacar(1500)
except SaldoInsuficienteError as e:
    print(f"❌ {e}")
except ValorInvalidoError as e:
    print(f"❌ {e}")
except Exception as e:
    print(f"❌ Erro inesperado: {e}")

🎯 Exemplo: Sistema de Validação

class EmailInvalidoError(Exception):
    """Erro quando email é inválido"""
    def __init__(self, email):
        self.email = email
        super().__init__(f"Email inválido: {email}")

class SenhaFracaError(Exception):
    """Erro quando senha é muito fraca"""
    def __init__(self, motivo):
        self.motivo = motivo
        super().__init__(f"Senha fraca: {motivo}")

class Usuario:
    def __init__(self, email, senha):
        self.validar_email(email)
        self.validar_senha(senha)
        self.email = email
        self.senha = senha
    
    def validar_email(self, email):
        if '@' not in email:
            raise EmailInvalidoError(email)
        if '.' not in email.split('@')[1]:
            raise EmailInvalidoError(email)
    
    def validar_senha(self, senha):
        if len(senha) < 8:
            raise SenhaFracaError("Senha deve ter pelo menos 8 caracteres")
        if not any(c.isdigit() for c in senha):
            raise SenhaFracaError("Senha deve conter pelo menos um número")

# Uso
try:
    usuario = Usuario("ana@email.com", "senha123")
except EmailInvalidoError as e:
    print(f"❌ {e}")
except SenhaFracaError as e:
    print(f"❌ {e}")
