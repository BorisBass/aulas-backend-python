Hierarquia de Exceções

Múltiplos except e ordem de captura

📊 Hierarquia de Exceções em Python

BaseException
    ├── Exception
    │   ├── ArithmeticError
    │   │   └── ZeroDivisionError
    │   ├── ValueError
    │   ├── TypeError
    │   ├── KeyError
    │   ├── IndexError
    │   └── ... (outras exceções)
    └── SystemExit
    └── KeyboardInterrupt

⚠️ Ordem Importa!

Python verifica os except na ordem que aparecem. Coloque exceções mais específicas primeiro!

try:
    # código que pode gerar exceções
    resultado = 10 / 0
except ZeroDivisionError:  # ✅ Específica primeiro
    print("Divisão por zero!")
except ArithmeticError:    # ✅ Depois a mais genérica
    print("Erro aritmético!")
except Exception:          # ✅ Por último, a mais genérica
    print("Erro genérico!")

# ❌ ERRADO - ordem invertida
try:
    resultado = 10 / 0
except Exception:          # ❌ Captura tudo primeiro!
    print("Erro genérico!")
except ZeroDivisionError:  # ❌ Nunca será executado!
    print("Divisão por zero!")

🏗️ Criando Hierarquia de Exceções Customizadas

# Exceção base
class ErroBancario(Exception):
    """Exceção base para erros bancários"""
    pass

# Exceções específicas que herdam da base
class SaldoInsuficienteError(ErroBancario):
    """Erro quando saldo é insuficiente"""
    pass

class ContaNaoEncontradaError(ErroBancario):
    """Erro quando conta não é encontrada"""
    pass

class LimiteExcedidoError(ErroBancario):
    """Erro quando limite é excedido"""
    pass

# Uso
try:
    # código que pode gerar erros bancários
    conta.sacar(1000)
except SaldoInsuficienteError:
    print("❌ Saldo insuficiente!")
except ContaNaoEncontradaError:
    print("❌ Conta não encontrada!")
except ErroBancario:  # Captura qualquer erro bancário
    print("❌ Erro bancário genérico!")
except Exception:
    print("❌ Erro inesperado!")

💡 Exemplo Completo: Sistema de Validação

# Hierarquia de exceções
class ErroValidacao(Exception):
    """Exceção base para erros de validação"""
    pass

class ErroFormato(ErroValidacao):
    """Erro de formato"""
    pass

class EmailInvalidoError(ErroFormato):
    """Email inválido"""
    pass

class TelefoneInvalidoError(ErroFormato):
    """Telefone inválido"""
    pass

class ErroRegraNegocio(ErroValidacao):
    """Erro de regra de negócio"""
    pass

class IdadeMinimaError(ErroRegraNegocio):
    """Idade mínima não atingida"""
    pass

# Uso com hierarquia
def validar_usuario(email, telefone, idade):
    try:
        if '@' not in email:
            raise EmailInvalidoError("Email inválido")
        if len(telefone) < 10:
            raise TelefoneInvalidoError("Telefone inválido")
        if idade < 18:
            raise IdadeMinimaError("Idade mínima: 18 anos")
        return True
    except EmailInvalidoError:
        print("❌ Erro: Email inválido")
    except TelefoneInvalidoError:
        print("❌ Erro: Telefone inválido")
    except ErroFormato:  # Captura EmailInvalidoError e TelefoneInvalidoError
        print("❌ Erro de formato")
    except IdadeMinimaError:
        print("❌ Erro: Idade mínima não atingida")
    except ErroValidacao:  # Captura todos os erros de validação
        print("❌ Erro de validação")
    except Exception:
        print("❌ Erro inesperado")

Atenção

Sempre coloque exceções mais específicas antes das mais genéricas! Python verifica na ordem e para no primeiro except que corresponder.
