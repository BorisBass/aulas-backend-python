BEP-022: Tratamento de Exceções em POO

Melhorando a segurança e robustez do código

Finally e Else

Blocos finally e else no tratamento de exceções

🔧 Bloco Finally

O bloco finally é sempre executado, independente de exceção ocorrer ou não. Útil para limpeza de recursos.

try:
    arquivo = open("dados.txt", "r")
    conteudo = arquivo.read()
    # Processar conteúdo...
except FileNotFoundError:
    print("❌ Arquivo não encontrado!")
finally:
    # SEMPRE executa, mesmo se houver erro
    arquivo.close()  # Garante que arquivo será fechado
    print("✅ Arquivo fechado")

# Exemplo com classe
class GerenciadorArquivo:
    def __init__(self, nome_arquivo):
        self.nome_arquivo = nome_arquivo
        self.arquivo = None
    
    def ler(self):
        try:
            self.arquivo = open(self.nome_arquivo, "r")
            return self.arquivo.read()
        except FileNotFoundError:
            print(f"❌ Arquivo {self.nome_arquivo} não encontrado!")
            return None
        finally:
            if self.arquivo:
                self.arquivo.close()
                print("✅ Arquivo fechado")

✅ Bloco Else

O bloco else é executado apenas se nenhuma exceção ocorrer.

try:
    numero = int(input("Digite um número: "))
    resultado = 10 / numero
except ValueError:
    print("❌ Valor inválido!")
except ZeroDivisionError:
    print("❌ Divisão por zero!")
else:
    # Executa APENAS se não houver exceção
    print(f"✅ Resultado: {resultado}")

# Exemplo prático
def processar_dados(dados):
    try:
        resultado = dados["valor"] / dados["divisor"]
    except KeyError:
        print("❌ Chave não encontrada!")
        return None
    except ZeroDivisionError:
        print("❌ Divisão por zero!")
        return None
    else:
        # Só executa se tudo der certo
        print(f"✅ Processamento concluído: {resultado}")
        return resultado

🔄 Estrutura Completa: Try-Except-Else-Finally

try:
    # Código que pode gerar exceção
    arquivo = open("dados.txt", "r")
    numero = int(arquivo.readline())
    resultado = 100 / numero
except FileNotFoundError:
    print("❌ Arquivo não encontrado!")
except ValueError:
    print("❌ Valor inválido no arquivo!")
except ZeroDivisionError:
    print("❌ Divisão por zero!")
else:
    # Executa se NENHUMA exceção ocorrer
    print(f"✅ Resultado calculado: {resultado}")
finally:
    # SEMPRE executa
    if 'arquivo' in locals():
        arquivo.close()
    print("✅ Limpeza concluída")

💡 Exemplo: Classe com Finally

class ConexaoBanco:
    def __init__(self):
        self.conectado = False
    
    def conectar(self):
        try:
            # Simula conexão
            self.conectado = True
            print("✅ Conectado ao banco")
        except Exception as e:
            print(f"❌ Erro na conexão: {e}")
        finally:
            # Sempre executa
            print("🔧 Tentativa de conexão finalizada")
    
    def executar_query(self, query):
        try:
            if not self.conectado:
                raise ConnectionError("Não conectado ao banco!")
            print(f"✅ Executando: {query}")
            return "Resultado da query"
        except ConnectionError as e:
            print(f"❌ {e}")
            return None
        finally:
            print("🔧 Query processada")

# Uso
conexao = ConexaoBanco()
conexao.conectar()
conexao.executar_query("SELECT * FROM usuarios")

Dica

Use finally para garantir que recursos sejam liberados (arquivos fechados, conexões encerradas, etc.), independente de erros. Use else para código que só deve executar quando tudo der certo!
