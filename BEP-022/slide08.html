BEP-022: Tratamento de Exceções em POO

Melhorando a segurança e robustez do código

Exemplo Completo: Sistema Bancário

Aplicando tratamento de exceções em um sistema real

# Exceções customizadas
class ErroBancario(Exception):
    """Exceção base para erros bancários"""
    pass

class SaldoInsuficienteError(ErroBancario):
    """Erro quando saldo é insuficiente"""
    def __init__(self, saldo, valor):
        self.saldo = saldo
        self.valor = valor
        super().__init__(
            f"Saldo insuficiente! Disponível: R$ {saldo:.2f}, "
            f"Solicitado: R$ {valor:.2f}"
        )

class ValorInvalidoError(ErroBancario):
    """Erro quando valor é inválido"""
    def __init__(self, valor):
        self.valor = valor
        super().__init__(f"Valor inválido: {valor}")

class ContaNaoEncontradaError(ErroBancario):
    """Erro quando conta não é encontrada"""
    def __init__(self, numero):
        self.numero = numero
        super().__init__(f"Conta {numero} não encontrada!")

# Classe principal
class ContaBancaria:
    def __init__(self, numero, titular, saldo_inicial=0):
        try:
            if saldo_inicial < 0:
                raise ValorInvalidoError(saldo_inicial)
            self.numero = numero
            self.titular = titular
            self.saldo = saldo_inicial
            print(f"✅ Conta {numero} criada para {titular}")
        except ValorInvalidoError as e:
            print(f"❌ Erro ao criar conta: {e}")
            raise
    
    def depositar(self, valor):
        try:
            if valor <= 0:
                raise ValorInvalidoError(valor)
            self.saldo += valor
            print(f"✅ Depósito de R$ {valor:.2f} realizado")
            return True
        except ValorInvalidoError as e:
            print(f"❌ Erro no depósito: {e}")
            return False
    
    def sacar(self, valor):
        try:
            if valor <= 0:
                raise ValorInvalidoError(valor)
            if valor > self.saldo:
                raise SaldoInsuficienteError(self.saldo, valor)
            self.saldo -= valor
            print(f"✅ Saque de R$ {valor:.2f} realizado")
            return True
        except (ValorInvalidoError, SaldoInsuficienteError) as e:
            print(f"❌ Erro no saque: {e}")
            return False
    
    def transferir(self, destino, valor):
        try:
            if not isinstance(destino, ContaBancaria):
                raise TypeError("Destino deve ser uma ContaBancaria")
            if self.sacar(valor):
                destino.depositar(valor)
                print(f"✅ Transferência de R$ {valor:.2f} realizada")
                return True
            return False
        except TypeError as e:
            print(f"❌ Erro na transferência: {e}")
            return False
        except Exception as e:
            print(f"❌ Erro inesperado: {e}")
            return False
    
    def consultar_saldo(self):
        return self.saldo

# Sistema de gerenciamento
class Banco:
    def __init__(self, nome):
        self.nome = nome
        self.contas = {}
    
    def criar_conta(self, numero, titular, saldo_inicial=0):
        try:
            if numero in self.contas:
                raise ValueError(f"Conta {numero} já existe!")
            conta = ContaBancaria(numero, titular, saldo_inicial)
            self.contas[numero] = conta
            return conta
        except (ValueError, ValorInvalidoError) as e:
            print(f"❌ Erro ao criar conta: {e}")
            return None
    
    def buscar_conta(self, numero):
        try:
            if numero not in self.contas:
                raise ContaNaoEncontradaError(numero)
            return self.contas[numero]
        except ContaNaoEncontradaError as e:
            print(f"❌ {e}")
            return None

# Uso do sistema
banco = Banco("Banco Python")

# Criar contas
conta1 = banco.criar_conta("001", "Ana", 1000)
conta2 = banco.criar_conta("002", "João", 500)

# Operações com tratamento de erros
conta1.sacar(1500)        # Erro: Saldo insuficiente
conta1.depositar(-100)    # Erro: Valor inválido
conta1.transferir(conta2, 200)  # Sucesso!

# Buscar conta
conta3 = banco.buscar_conta("999")  # Erro: Conta não encontrada
