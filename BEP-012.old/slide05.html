Filtros e Ordenação

WHERE, ORDER BY e LIMIT

WHERE

SELECT id, nome FROM clientes
WHERE email IS NOT NULL AND nome LIKE 'A%';

ORDER BY

SELECT id, nome FROM clientes
ORDER BY nome ASC;

LIMIT

SELECT * FROM pedidos
ORDER BY criado_em DESC
LIMIT 5;
