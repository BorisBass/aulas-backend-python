DDL: CREATE, ALTER, DROP

Definição e evolução do esquema relacional

CREATE

CREATE TABLE clientes (
  id INTEGER PRIMARY KEY,
  nome TEXT NOT NULL,
  email TEXT UNIQUE,
  criado_em DATETIME DEFAULT CURRENT_TIMESTAMP
);

ALTER

-- Adicionando coluna
ALTER TABLE clientes ADD COLUMN ativo INTEGER DEFAULT 1;

-- Renomeando tabela (SQLite 3.25+)
ALTER TABLE clientes RENAME TO clientes_v2;

DROP

DROP TABLE IF EXISTS clientes_v2;

Atenção: DROP é destrutivo. Sempre faça backup e confirme escopo.
