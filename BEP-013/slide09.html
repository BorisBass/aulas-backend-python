Preparação do Ambiente

Primeiro, vamos criar as tabelas e inserir dados de exemplo:

                        

                            -- 1. Criar tabela clientes
CREATE TABLE clientes (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT NOT NULL,
    email TEXT UNIQUE,
    telefone TEXT,
    cidade TEXT,
    idade INTEGER,
    categoria TEXT DEFAULT 'Normal'
);
                        

                            -- 2. Criar tabela pedidos
CREATE TABLE pedidos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    cliente_id INTEGER,
    produto TEXT,
    valor REAL,
    data_pedido DATE,
    status TEXT DEFAULT 'pendente',
    FOREIGN KEY (cliente_id) REFERENCES clientes(id)
);

Exercício 1: Remoção por ID

Objetivo: Remover o cliente com ID 5 (verificar se não tem pedidos primeiro)

                        

                            -- 1. Verificar se cliente tem pedidos
SELECT COUNT(*) FROM pedidos WHERE cliente_id = 5;
-- 2. Se não tiver pedidos, deletar cliente
DELETE FROM clientes WHERE id = 5;
-- 3. Verificar resultado
SELECT * FROM clientes WHERE id = 5;

Exercício 2: Remoção por Condição

Objetivo: Remover todos os clientes sem telefone

                        

                            -- 1. Verificar quantos clientes serão removidos
SELECT COUNT(*) FROM clientes WHERE telefone IS NULL;
-- 2. Ver quais clientes serão removidos
SELECT * FROM clientes WHERE telefone IS NULL;
-- 3. Executar DELETE
DELETE FROM clientes WHERE telefone IS NULL;
-- 4. Verificar resultado
SELECT COUNT(*) FROM clientes;

Exercício 3: Remoção por Data

Objetivo: Remover pedidos antigos (anteriores a 2020)

                        

                            -- 1. Verificar quantos pedidos serão removidos
SELECT COUNT(*) FROM pedidos WHERE data_pedido < '2020-01-01';
-- 2. Ver quais pedidos serão removidos
SELECT * FROM pedidos WHERE data_pedido < '2020-01-01';
-- 3. Executar DELETE
DELETE FROM pedidos WHERE data_pedido < '2020-01-01';
-- 4. Verificar resultado
SELECT COUNT(*) FROM pedidos;

Exercício 4: Remoção por Status

Objetivo: Remover todos os pedidos cancelados

                        

                            -- 1. Verificar quantos pedidos cancelados existem
SELECT COUNT(*) FROM pedidos WHERE status = 'cancelado';
-- 2. Ver quais pedidos serão removidos
SELECT * FROM pedidos WHERE status = 'cancelado';
-- 3. Executar DELETE
DELETE FROM pedidos WHERE status = 'cancelado';
-- 4. Verificar resultado
SELECT status, COUNT(*) FROM pedidos GROUP BY status;

Exercício 5: Remoção com Múltiplas Condições

Objetivo: Remover pedidos cancelados de 2019

                        

                            -- 1. Verificar quantos pedidos serão removidos
SELECT COUNT(*) FROM pedidos
WHERE status = 'cancelado' AND data_pedido BETWEEN '2019-01-01' AND '2019-12-31';
-- 2. Ver quais pedidos serão removidos
SELECT * FROM pedidos
WHERE status = 'cancelado' AND data_pedido BETWEEN '2019-01-01' AND '2019-12-31';
-- 3. Executar DELETE
DELETE FROM pedidos
WHERE status = 'cancelado' AND data_pedido BETWEEN '2019-01-01' AND '2019-12-31';
-- 4. Verificar resultado
SELECT COUNT(*) FROM pedidos;
