UPDATE, DELETE e boas práticas para modificação de dados

Sistema CRUD Completo

Sistema de Gerenciamento de Clientes

Vamos criar um sistema completo em Python que implementa todas as operações CRUD (Create, Read, Update, Delete) com SQLite3. Este sistema demonstra como usar UPDATE e DELETE em um contexto real.

Código Completo do Sistema

import sqlite3
import os
from datetime import datetime
# Classe para gerenciar o banco de dados
class GerenciadorClientes:
    def __init__(self, nome_banco='clientes.db'):
        self.nome_banco = nome_banco
        self.conectar()
        self.criar_tabela()

Métodos de Conexão e Criação

    def conectar(self):
        self.conn = sqlite3.connect(self.nome_banco)
        self.cursor = self.conn.cursor()
        print(f"Conectado ao banco: {self.nome_banco}")
    def criar_tabela(self):
        self.cursor.execute('''
CREATE TABLE IF NOT EXISTS clientes (
id INTEGER PRIMARY KEY AUTOINCREMENT,
nome TEXT NOT NULL,
email TEXT UNIQUE,
telefone TEXT,
cidade TEXT,
idade INTEGER,
categoria TEXT DEFAULT 'Normal',
data_cadastro TIMESTAMP DEFAULT CURRENT_TIMESTAMP
)''')
        self.conn.commit()

Operações CREATE e READ

    def adicionar_cliente(self, nome, email, telefone, cidade, idade):
        try:
            self.cursor.execute('''
INSERT INTO clientes (nome, email, telefone, cidade, idade)
VALUES (?, ?, ?, ?, ?)''', (nome, email, telefone, cidade, idade))
            self.conn.commit()
            print(f"Cliente {nome} adicionado com sucesso!")
        except sqlite3.IntegrityError:
            print("Erro: Email já existe!")
    def listar_clientes(self):
        self.cursor.execute("SELECT * FROM clientes ORDER BY nome")
        return self.cursor.fetchall()

Operações UPDATE

    def atualizar_cliente(self, cliente_id, campo, novo_valor):
        # Verificar se cliente existe
        self.cursor.execute("SELECT nome FROM clientes WHERE id = ?", (cliente_id,))
        if not self.cursor.fetchone():
            print("Cliente não encontrado!")
            return
        # Executar UPDATE
        self.cursor.execute(f"UPDATE clientes SET {campo} = ? WHERE id = ?", (novo_valor, cliente_id))
        self.conn.commit()
        print(f"Cliente {cliente_id} atualizado com sucesso!")

Operações DELETE

    def remover_cliente(self, cliente_id):
        # Verificar se cliente existe
        self.cursor.execute("SELECT nome FROM clientes WHERE id = ?", (cliente_id,))
        cliente = self.cursor.fetchone()
        if not cliente:
            print("Cliente não encontrado!")
            return
        # Confirmar remoção
        confirmacao = input(f"Tem certeza que deseja remover {cliente[0]}? (s/n): ")
        if confirmacao.lower() == 's':
            self.cursor.execute("DELETE FROM clientes WHERE id = ?", (cliente_id,))
            self.conn.commit()
            print(f"Cliente {cliente[0]} removido com sucesso!")
        else:
            print("Operação cancelada.")

Métodos de Busca

    def buscar_cliente(self, termo):
        self.cursor.execute('''
SELECT * FROM clientes
WHERE nome LIKE ? OR email LIKE ? OR cidade LIKE ?''',
            (f'%{termo}%', f'%{termo}%', f'%{termo}%'))
        return self.cursor.fetchall()
    def fechar_conexao(self):
        self.conn.close()
        print("Conexão fechada.")

Menu Interativo

def menu_principal():
    gerenciador = GerenciadorClientes()
    while True:
        print("\n=== SISTEMA DE GERENCIAMENTO DE CLIENTES ===")
        print("1. Adicionar cliente")
        print("2. Listar clientes")
        print("3. Atualizar cliente")
        print("4. Remover cliente")
        print("5. Buscar cliente")
        print("0. Sair")
        opcao = input("\nEscolha uma opção: ")
        if opcao == '0':
            gerenciador.fechar_conexao()
            break
        elif opcao == '1':
            nome = input("Nome: ")
            email = input("Email: ")
            telefone = input("Telefone: ")
            cidade = input("Cidade: ")
            idade = int(input("Idade: "))
            gerenciador.adicionar_cliente(nome, email, telefone, cidade, idade)
        elif opcao == '2':
            clientes = gerenciador.listar_clientes()
            for cliente in clientes:
                print(f"ID: {cliente[0]}, Nome: {cliente[1]}, Email: {cliente[2]}, Cidade: {cliente[4]}")
        elif opcao == '3':
            cliente_id = int(input("ID do cliente: "))
            campo = input("Campo a atualizar (nome/email/telefone/cidade/idade/categoria): ")
            novo_valor = input("Novo valor: ")
            gerenciador.atualizar_cliente(cliente_id, campo, novo_valor)
        elif opcao == '4':
            cliente_id = int(input("ID do cliente a remover: "))
            gerenciador.remover_cliente(cliente_id)
        elif opcao == '5':
            termo = input("Termo de busca: ")
            resultados = gerenciador.buscar_cliente(termo)
            for cliente in resultados:
                print(f"ID: {cliente[0]}, Nome: {cliente[1]}, Email: {cliente[2]}")
        else:
            print("Opção inválida!")
# Executar o programa
if __name__ == "__main__":
    menu_principal()

Conceitos Aplicados

  • UPDATE com WHERE: Atualização segura de registros específicos
  • DELETE com confirmação: Remoção segura com validação
  • Transações: Operações atômicas com commit/rollback
  • Validação de dados: Verificação antes de operações
  • Tratamento de erros: Try/except para operações críticas
  • Interface amigável: Menu interativo para o usuário
