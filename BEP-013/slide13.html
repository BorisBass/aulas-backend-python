7. UPDATE - Atualizar Cliente

Conceito: Função para atualizar informações de clientes existentes.

Prática: Validação de ID e atualização segura com WHERE.

Célula 7: Atualizar Cliente

def atualizar_cliente(cursor, conn):
    """Atualiza informações de um cliente"""
    try:
        print("\n✏️ ATUALIZAR CLIENTE")
        print("-" * 30)
        
        # Listar clientes para facilitar a escolha
        cursor.execute("SELECT id, nome FROM clientes ORDER BY nome")
        clientes = cursor.fetchall()
        
        if not clientes:
            print("📭 Nenhum cliente cadastrado!")
            return
        
        print("👥 Clientes disponíveis:")
        for id_cliente, nome in clientes:
            print(f"  {id_cliente} - {nome}")
        
        # Solicitar ID do cliente
        try:
            cliente_id = int(input("\n👉 Digite o ID do cliente: "))
        except ValueError:
            print("❌ ID deve ser um número!")
            return
        
        # Verificar se cliente existe
        cursor.execute("SELECT * FROM clientes WHERE id = ?", (cliente_id,))
        cliente = cursor.fetchone()
        
        if not cliente:
            print("❌ Cliente não encontrado!")
            return
        
        # Mostrar dados atuais
        print(f"\n📋 Dados atuais do cliente {cliente[1]}:")
        print(f"  Nome: {cliente[1]}")
        print(f"  Email: {cliente[2] if cliente[2] else 'Não informado'}")
        print(f"  Telefone: {cliente[3] if cliente[3] else 'Não informado'}")
        print(f"  Cidade: {cliente[4] if cliente[4] else 'Não informado'}")
        
        # Solicitar novos dados
        print("\n✏️ Digite os novos dados (pressione Enter para manter o atual):")
        
        novo_nome = input(f"Nome [{cliente[1]}]: ").strip()
        if not novo_nome:
            novo_nome = cliente[1]
        
        novo_email = input(f"Email [{cliente[2] if cliente[2] else 'vazio'}]: ").strip()
        if not novo_email:
            novo_email = cliente[2]
        
        novo_telefone = input(f"Telefone [{cliente[3] if cliente[3] else 'vazio'}]: ").strip()
        if not novo_telefone:
            novo_telefone = cliente[3]
        
        novo_cidade = input(f"Cidade [{cliente[4] if cliente[4] else 'vazio'}]: ").strip()
        if not novo_cidade:
            novo_cidade = cliente[4]
        
        # Atualizar no banco
        cursor.execute('''
            UPDATE clientes 
            SET nome = ?, email = ?, telefone = ?, cidade = ?
            WHERE id = ?
        ''', (novo_nome, novo_email, novo_telefone, novo_cidade, cliente_id))
        
        conn.commit()
        print(f"✅ Cliente '{novo_nome}' atualizado com sucesso!")
        
    except sqlite3.IntegrityError:
        print("❌ Erro: Email já cadastrado em outro cliente!")
    except sqlite3.Error as e:
        print(f"❌ Erro no banco: {e}")
    except Exception as e:
        print(f"❌ Erro inesperado: {e}")

8. DELETE - Remover Cliente

Conceito: Função para remover clientes com confirmação de segurança.

Prática: Verificação de dependências e confirmação antes de remover.

Célula 8: Remover Cliente

def remover_cliente(cursor, conn):
    """Remove um cliente do banco"""
    try:
        print("\n🗑️ REMOVER CLIENTE")
        print("-" * 30)
        
        # Listar clientes para facilitar a escolha
        cursor.execute("SELECT id, nome FROM clientes ORDER BY nome")
        clientes = cursor.fetchall()
        
        if not clientes:
            print("📭 Nenhum cliente cadastrado!")
            return
        
        print("👥 Clientes disponíveis:")
        for id_cliente, nome in clientes:
            print(f"  {id_cliente} - {nome}")
        
        # Solicitar ID do cliente
        try:
            cliente_id = int(input("\n👉 Digite o ID do cliente a remover: "))
        except ValueError:
            print("❌ ID deve ser um número!")
            return
        
        # Verificar se cliente existe
        cursor.execute("SELECT * FROM clientes WHERE id = ?", (cliente_id,))
        cliente = cursor.fetchone()
        
        if not cliente:
            print("❌ Cliente não encontrado!")
            return
        
        # Mostrar dados do cliente
        print(f"\n📋 Dados do cliente a ser removido:")
        print(f"  ID: {cliente[0]}")
        print(f"  Nome: {cliente[1]}")
        print(f"  Email: {cliente[2] if cliente[2] else 'Não informado'}")
        print(f"  Telefone: {cliente[3] if cliente[3] else 'Não informado'}")
        print(f"  Cidade: {cliente[4] if cliente[4] else 'Não informado'}")
        print(f"  Data de cadastro: {cliente[5]}")
        
        # Verificar se tem pedidos (se a tabela existir)
        try:
            cursor.execute("SELECT COUNT(*) FROM pedidos WHERE cliente_id = ?", (cliente_id,))
            pedidos_count = cursor.fetchone()[0]
            if pedidos_count > 0:
                print(f"\n⚠️ ATENÇÃO: Este cliente tem {pedidos_count} pedido(s) associado(s)!")
                print("   Remover o cliente pode causar inconsistências nos dados.")
        except sqlite3.Error:
            # Tabela pedidos não existe, continuar normalmente
            pass
        
        # Confirmação de segurança
        print(f"\n⚠️ ATENÇÃO: Esta ação é IRREVERSÍVEL!")
        confirmacao = input(f"Tem certeza que deseja remover '{cliente[1]}'? (digite 'SIM' para confirmar): ")
        
        if confirmacao.upper() == 'SIM':
            # Executar remoção
            cursor.execute("DELETE FROM clientes WHERE id = ?", (cliente_id,))
            conn.commit()
            print(f"✅ Cliente '{cliente[1]}' removido com sucesso!")
        else:
            print("❌ Operação cancelada pelo usuário.")
        
    except sqlite3.Error as e:
        print(f"❌ Erro no banco: {e}")
    except Exception as e:
        print(f"❌ Erro inesperado: {e}")

def exibir_menu_completo():
    """Exibe o menu principal completo com CRUD"""

    print("="*50)

    print("2. 📋 Listar todos os clientes (READ)")
    print("3. 🔍 Buscar cliente por nome (READ)")
    print("4. 📊 Estatísticas (READ)")
    print("5. ✏️ Atualizar cliente (UPDATE)")
    print("6. 🗑️ Remover cliente (DELETE)")
    print("0. 🚪 Sair")
    print("="*50)

def main_completo():
    """Função principal do programa com CRUD completo"""
    print("🚀 Iniciando Sistema de Cadastro de Clientes - CRUD Completo...")
    
    # Conectar ao banco
    conn, cursor = conectar_banco()
    if not conn:
        print("❌ Não foi possível conectar ao banco. Encerrando...")
        return
    
    try:
        while True:
            exibir_menu_completo()
            opcao = input("👉 Escolha uma opção: ").strip()
            
            if opcao == '1':
                cadastrar_cliente(cursor, conn)
            elif opcao == '2':
                listar_clientes(cursor)
            elif opcao == '3':
                buscar_cliente(cursor)
            elif opcao == '4':
                mostrar_estatisticas(cursor)
            elif opcao == '5':
                atualizar_cliente(cursor, conn)
            elif opcao == '6':
                remover_cliente(cursor, conn)
            elif opcao == '0':
                print("\n👋 Obrigado por usar o sistema! Até logo!")
                break
            else:
                print("❌ Opção inválida! Tente novamente.")
            
            input("\n⏸️ Pressione Enter para continuar...")
    
    except KeyboardInterrupt:
        print("\n\n👋 Programa interrompido pelo usuário.")
    except Exception as e:
        print(f"\n❌ Erro inesperado: {e}")
    finally:
        # Sempre fechar a conexão
        if conn:
            conn.close()
            print("🔒 Conexão com banco fechada.")

# Executar o programa completo
if __name__ == "__main__":
    main_completo()

Conceitos Aplicados - CRUD Completo

  • CREATE: INSERT para cadastrar novos clientes
  • READ: SELECT para listar, buscar e gerar estatísticas
  • UPDATE: UPDATE com WHERE para modificar dados existentes
  • DELETE: DELETE com WHERE para remover registros
  • Validação de Dados: Verificação de entrada e existência de registros
  • Confirmação de Segurança: Proteção contra remoções acidentais
  • Tratamento de Erros: Try/except para todas as operações
  • Transações: Commit para salvar alterações
  • Interface Amigável: Menu interativo e mensagens claras
  • Verificação de Dependências: Checagem antes de remover

Boas Práticas Implementadas

  • SEMPRE usar WHERE: Todas as operações UPDATE e DELETE têm WHERE
  • Verificar antes de modificar: SELECT antes de UPDATE/DELETE
  • Confirmação de segurança: Usuário deve confirmar remoções
  • Tratamento de erros: Try/except em todas as operações
  • Validação de entrada: Verificação de dados do usuário
  • Transações seguras: Commit apenas após validação
  • Mensagens claras: Feedback para o usuário
  • Verificação de dependências: Checagem de relacionamentos
