1) Classe Retangulo

  1. Atributos: largura, altura.
  2. Métodos: area() e perimetro().
  3. Crie dois objetos e compare as áreas.
class Retangulo:
    def __init__(self, largura, altura):
        self.largura = float(largura)
        self.altura = float(altura)

    def area(self):
        return self.largura * self.altura

    def perimetro(self):
        return 2 * (self.largura + self.altura)

r1 = Retangulo(3, 4)
r2 = Retangulo(5, 2)
print(r1.area(), r1.perimetro())
print(r2.area(), r2.perimetro())

class Livro:

3) Desafio: MediaAluno

  1. Atributos: nome, notas (lista de floats).
  2. Método: media() retorna média simples.
  3. Método: situacao() retorna "Aprovado" se média >= 6, senão "Recuperação".
class MediaAluno:
    def __init__(self, nome, notas):
        self.nome = nome
        self.notas = [float(n) for n in notas]

    def media(self):
        return sum(self.notas) / len(self.notas)

    def situacao(self):
        return "Aprovado" if self.media() >= 6 else "Recuperação"

m = MediaAluno("Ana", [7, 6, 8])
print(m.media(), m.situacao())
