Comandos SQL Essenciais

Comandos Básicos

# Conectar ao banco
conn = sqlite3.connect('banco.db')
cursor = conn.cursor()

# Criar tabela
cursor.execute('''
    CREATE TABLE IF NOT EXISTS tabela (
        id INTEGER PRIMARY KEY,
        nome TEXT NOT NULL
    )
''')

# Inserir dados
cursor.execute('''
    INSERT INTO tabela (nome)
    VALUES (?)
''', ('valor',))

# Consultar dados
cursor.execute("SELECT * FROM tabela")
dados = cursor.fetchall()

# Atualizar dados
cursor.execute('''
    UPDATE tabela 
    SET nome = ? 
    WHERE id = ?
''', ('novo_valor', 1))

# Deletar dados
cursor.execute("DELETE FROM tabela WHERE id = ?", (1,))

# Salvar alterações
conn.commit()

# Fechar conexão
conn.close()
