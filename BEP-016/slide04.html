Consultas com Múltiplas Tabelas

Em sistemas reais, frequentemente precisamos trabalhar com dados de várias tabelas relacionadas. Vamos ver como fazer isso com SQLite.

Exemplo: Sistema de Biblioteca

Vamos criar um sistema com duas tabelas relacionadas:

Tabela Livros

  • id (chave primária)
  • titulo
  • autor
  • ano_publicacao

Tabela Emprestimos

  • id (chave primária)
  • livro_id (chave estrangeira)
  • usuario
  • data_emprestimo
  • data_devolucao

Criando Tabelas Relacionadas

import sqlite3

def criar_banco_biblioteca():
    """Cria o banco da biblioteca com tabelas relacionadas"""
    conn = sqlite3.connect('biblioteca.db')
    cursor = conn.cursor()
    
    # Criar tabela de livros
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS livros (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            titulo TEXT NOT NULL,
            autor TEXT NOT NULL,
            ano_publicacao INTEGER,
            disponivel BOOLEAN DEFAULT 1
        )
    ''')
    
    # Criar tabela de empréstimos
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS emprestimos (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            livro_id INTEGER NOT NULL,
            usuario TEXT NOT NULL,
            data_emprestimo DATE DEFAULT CURRENT_DATE,
            data_devolucao DATE,
            FOREIGN KEY (livro_id) REFERENCES livros (id)
        )
    ''')
    
    conn.commit()
    print("✅ Banco da biblioteca criado!")
    
    # Inserir dados de exemplo
    livros_exemplo = [
        ('Python para Iniciantes', 'João Silva', 2023),
        ('Banco de Dados', 'Maria Santos', 2022),
        ('Programação Web', 'Pedro Costa', 2024)
    ]
    
    cursor.executemany('''
        INSERT INTO livros (titulo, autor, ano_publicacao)
        VALUES (?, ?, ?)
    ''', livros_exemplo)
    
    conn.commit()
    print("✅ Livros de exemplo inseridos!")
    
    conn.close()

JOIN - Unindo Tabelas

Conceito: JOIN permite combinar dados de duas ou mais tabelas baseado em uma relação.

Consultas com JOIN

import sqlite3

def consultas_join():
    """Exemplos de consultas com JOIN"""
    conn = sqlite3.connect('biblioteca.db')
    cursor = conn.cursor()
    
    print("📚 LIVROS E SEUS EMPRÉSTIMOS")
    print("-" * 60)
    
    # INNER JOIN - mostra apenas livros que têm empréstimos
    cursor.execute('''
        SELECT l.titulo, l.autor, e.usuario, e.data_emprestimo
        FROM livros l
        INNER JOIN emprestimos e ON l.id = e.livro_id
        ORDER BY e.data_emprestimo DESC
    ''')
    
    emprestimos = cursor.fetchall()
    for emprestimo in emprestimos:
        print(f"Livro: {emprestimo[0]} | Autor: {emprestimo[1]} | Usuário: {emprestimo[2]} | Data: {emprestimo[3]}")
    
    print("\n📖 TODOS OS LIVROS (COM OU SEM EMPRÉSTIMOS)")
    print("-" * 60)
    
    # LEFT JOIN - mostra todos os livros, mesmo sem empréstimos
    cursor.execute('''
        SELECT l.titulo, l.autor, 
               CASE 
                   WHEN e.usuario IS NOT NULL THEN e.usuario
                   ELSE 'Disponível'
               END as status
        FROM livros l
        LEFT JOIN emprestimos e ON l.id = e.livro_id AND e.data_devolucao IS NULL
        ORDER BY l.titulo
    ''')
    
    livros_status = cursor.fetchall()
    for livro in livros_status:
        print(f"Livro: {livro[0]} | Autor: {livro[1]} | Status: {livro[2]}")
    
    conn.close()

Consultas Agregadas

Conceito: Funções como COUNT, SUM, AVG, MAX, MIN para análise de dados.

Estatísticas da Biblioteca

def estatisticas_biblioteca():
    """Gera estatísticas da biblioteca"""
    conn = sqlite3.connect('biblioteca.db')
    cursor = conn.cursor()
    
    print("📊 ESTATÍSTICAS DA BIBLIOTECA")
    print("=" * 40)
    
    # Total de livros
    cursor.execute("SELECT COUNT(*) FROM livros")
    total_livros = cursor.fetchone()[0]
    print(f"📚 Total de livros: {total_livros}")
    
    # Livros por autor
    cursor.execute('''
        SELECT autor, COUNT(*) as quantidade
        FROM livros
        GROUP BY autor
        ORDER BY quantidade DESC
    ''')
    
    autores = cursor.fetchall()
    print(f"\n👨‍💼 Livros por autor:")
    for autor, qtd in autores:
        print(f"  {autor}: {qtd} livro(s)")
    
    # Livros mais recentes
    cursor.execute('''
        SELECT titulo, ano_publicacao
        FROM livros
        ORDER BY ano_publicacao DESC
        LIMIT 3
    ''')
    
    recentes = cursor.fetchall()
    print(f"\n🆕 Livros mais recentes:")
    for livro in recentes:
        print(f"  {livro[0]} ({livro[1]})")
    
    # Total de empréstimos
    cursor.execute("SELECT COUNT(*) FROM emprestimos")
    total_emprestimos = cursor.fetchone()[0]
    print(f"\n📖 Total de empréstimos: {total_emprestimos}")
    
    conn.close()

Consultas com Filtros Complexos

Conceito: WHERE com múltiplas condições e subconsultas.

Filtros Avançados

def consultas_filtradas():
    """Exemplos de consultas com filtros"""
    conn = sqlite3.connect('biblioteca.db')
    cursor = conn.cursor()
    
    print("🔍 CONSULTAS FILTRADAS")
    print("=" * 40)
    
    # Livros de um autor específico
    autor_busca = input("Digite o nome do autor: ").strip()
    cursor.execute('''
        SELECT titulo, ano_publicacao
        FROM livros
        WHERE autor LIKE ?
        ORDER BY ano_publicacao DESC
    ''', (f"%{autor_busca}%",))
    
    livros_autor = cursor.fetchall()
    print(f"\n📚 Livros do autor '{autor_busca}':")
    for livro in livros_autor:
        print(f"  {livro[0]} ({livro[1]})")
    
    # Livros disponíveis (não emprestados)
    cursor.execute('''
        SELECT l.titulo, l.autor
        FROM livros l
        LEFT JOIN emprestimos e ON l.id = e.livro_id AND e.data_devolucao IS NULL
        WHERE e.id IS NULL
        ORDER BY l.titulo
    ''')
    
    disponiveis = cursor.fetchall()
    print(f"\n✅ Livros disponíveis:")
    for livro in disponiveis:
        print(f"  {livro[0]} - {livro[1]}")
    
    # Usuários que mais emprestam
    cursor.execute('''
        SELECT usuario, COUNT(*) as total_emprestimos
        FROM emprestimos
        GROUP BY usuario
        ORDER BY total_emprestimos DESC
        LIMIT 5
    ''')
    
    usuarios_ativos = cursor.fetchall()
    print(f"\n👥 Usuários mais ativos:")
    for usuario, total in usuarios_ativos:
        print(f"  {usuario}: {total} empréstimo(s)")
    
    conn.close()

Conceitos Importantes

  • Chave Estrangeira: Campo que referencia outra tabela
  • INNER JOIN: Apenas registros que têm correspondência
  • LEFT JOIN: Todos os registros da tabela à esquerda
  • GROUP BY: Agrupa registros para funções agregadas
  • ORDER BY: Ordena os resultados
  • LIMIT: Limita o número de resultados
