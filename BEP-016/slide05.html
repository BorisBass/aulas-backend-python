Exercício 1: Sistema de Produtos

Crie um sistema para gerenciar produtos de uma loja com as seguintes funcionalidades:

Estrutura da Tabela

# Tabela produtos:
# - id (INTEGER PRIMARY KEY)
# - nome (TEXT NOT NULL)
# - preco (REAL NOT NULL)
# - categoria (TEXT)
# - estoque (INTEGER DEFAULT 0)
# - data_cadastro (DATE DEFAULT CURRENT_DATE)

import sqlite3

def criar_tabela_produtos():
    """Cria a tabela de produtos"""
    conn = sqlite3.connect('loja.db')
    cursor = conn.cursor()
    
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS produtos (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            nome TEXT NOT NULL,
            preco REAL NOT NULL,
            categoria TEXT,
            estoque INTEGER DEFAULT 0,
            data_cadastro DATE DEFAULT CURRENT_DATE
        )
    ''')
    
    conn.commit()
    print("✅ Tabela de produtos criada!")
    conn.close()

Funcionalidades a Implementar

Cadastrar Produto

  • Nome obrigatório
  • Preço obrigatório
  • Categoria opcional
  • Estoque inicial

Listar Produtos

  • Mostrar todos os produtos
  • Ordenar por nome
  • Formatação bonita

Buscar Produto

  • Buscar por nome
  • Buscar por categoria
  • Mostrar resultados

Estatísticas

  • Total de produtos
  • Produtos por categoria
  • Valor total do estoque

Exercício 2: Sistema de Contatos

Crie um sistema de agenda de contatos com funcionalidades avançadas:

Estrutura Avançada

# Tabela contatos:
# - id (INTEGER PRIMARY KEY)
# - nome (TEXT NOT NULL)
# - telefone (TEXT UNIQUE)
# - email (TEXT UNIQUE)
# - cidade (TEXT)
# - data_nascimento (DATE)
# - data_cadastro (TIMESTAMP DEFAULT CURRENT_TIMESTAMP)

import sqlite3
from datetime import datetime

def criar_sistema_contatos():
    """Cria o sistema de contatos"""
    conn = sqlite3.connect('contatos.db')
    cursor = conn.cursor()
    
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS contatos (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            nome TEXT NOT NULL,
            telefone TEXT UNIQUE,
            email TEXT UNIQUE,
            cidade TEXT,
            data_nascimento DATE,
            data_cadastro TIMESTAMP DEFAULT CURRENT_TIMESTAMP

    ''')
    
    conn.commit()
    print("✅ Sistema de contatos criado!")
    conn.close()

Exercício 3: Sistema de Notas

Crie um sistema para gerenciar notas de alunos com múltiplas disciplinas:

Estrutura com Relacionamentos

# Tabela alunos:
# - id (INTEGER PRIMARY KEY)
# - nome (TEXT NOT NULL)
# - matricula (TEXT UNIQUE)
# - curso (TEXT)

# Tabela disciplinas:
# - id (INTEGER PRIMARY KEY)
# - nome (TEXT NOT NULL)
# - codigo (TEXT UNIQUE)

# Tabela notas:
# - id (INTEGER PRIMARY KEY)
# - aluno_id (INTEGER)
# - disciplina_id (INTEGER)
# - nota (REAL)
# - data_avaliacao (DATE)

import sqlite3

def criar_sistema_notas():
    """Cria o sistema de notas"""
    conn = sqlite3.connect('notas.db')
    cursor = conn.cursor()
    
    # Criar tabela de alunos
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS alunos (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            nome TEXT NOT NULL,
            matricula TEXT UNIQUE,
            curso TEXT
        )
    ''')
    
    # Criar tabela de disciplinas
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS disciplinas (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            nome TEXT NOT NULL,
            codigo TEXT UNIQUE
        )
    ''')
    
    # Criar tabela de notas
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS notas (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            aluno_id INTEGER,
            disciplina_id INTEGER,
            nota REAL,
            data_avaliacao DATE DEFAULT CURRENT_DATE,
            FOREIGN KEY (aluno_id) REFERENCES alunos (id),
            FOREIGN KEY (disciplina_id) REFERENCES disciplinas (id)
        )
    ''')
    
    conn.commit()
    print("✅ Sistema de notas criado!")
    conn.close()
