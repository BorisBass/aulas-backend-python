Método format()

🔧 Método format()

Formatação de strings usando o método format()

# Formatação posicional
nome = "Maria"
idade = 30
print("Olá, {}! Você tem {} anos.".format(nome, idade))

# Formatação com índices
print("Olá, {0}! Você tem {1} anos.".format(nome, idade))
print("Idade: {1}, Nome: {0}".format(nome, idade))

🏷️ Formatação com Nomes

# Formatação com nomes
print("Olá, {nome}! Você tem {idade} anos.".format(nome="João", idade=25))

# Com dicionário
dados = {"nome": "Ana", "idade": 28}
print("Olá, {nome}! Você tem {idade} anos.".format(**dados))

📊 Formatação de Números

# Formatação de decimais
preco = 19.99
print("Preço: R$ {:.2f}".format(preco))

# Formatação com separadores
numero = 1000000
print("Número: {:,}".format(numero))  # 1,000,000

# Formatação de percentual
taxa = 0.15
print("Taxa: {:.1%}".format(taxa))
💡 Exemplo Prático
# Relatório de vendas
vendas = [
    {"produto": "Notebook", "quantidade": 5, "preco": 2500.00},
    {"produto": "Mouse", "quantidade": 10, "preco": 25.50}
]

for item in vendas:
    total = item["quantidade"] * item["preco"]
    print("{produto:10} | {quantidade:2d} x R$ {preco:7.2f} = R$ {total:8.2f}".format(
        produto=item["produto"],
        quantidade=item["quantidade"],
        preco=item["preco"],
        total=total
    ))
