Prática com classes encapsuladas

1) Classe Aluno Encapsulada (com @property)

  1. Crie a classe Aluno com atributo privado __nota.
  2. Implemente @property e @nota.setter com validação (0-10).
  3. Adicione método aprovado() que retorna True se nota >= 6.
class Aluno:
    def __init__(self, nome, nota):
        self.nome = nome
        self.__nota = 0
        self.nota = nota  # Usa o setter
    
    @property
    def nota(self):
        return self.__nota
    
    @nota.setter
    def nota(self, valor):
        if 0 <= valor <= 10:
            self.__nota = valor
        else:
            raise ValueError("Nota deve estar entre 0 e 10")
    
    def aprovado(self):
        return self.__nota >= 6

a = Aluno("Ana", 7.5)
print(f"Nota: {a.nota}, Aprovado: {a.aprovado()}")  # Sintaxe elegante
a.nota = 8.0  # Atribuição direta com validação
print(f"Nova nota: {a.nota}")

2) Classe Produto Encapsulada (com @property)

  1. Crie a classe Produto com __preco e __estoque privados.
  2. Implemente @property e setters para ambos os atributos.
  3. Valide: preço > 0 e estoque >= 0.
  4. Adicione método vender(quantidade) que reduz estoque.
class Produto:
    def __init__(self, nome, preco, estoque):
        self.nome = nome
        self.__preco = 0
        self.__estoque = 0
        self.preco = preco  # Usa o setter
        self.estoque = estoque  # Usa o setter
    
    @property
    def preco(self):
        return self.__preco
    
    @preco.setter
    def preco(self, valor):
        if valor > 0:
            self.__preco = float(valor)
        else:
            raise ValueError("Preço deve ser maior que zero")
    
    @property
    def estoque(self):
        return self.__estoque
    
    @estoque.setter
    def estoque(self, valor):
        if valor >= 0:
            self.__estoque = int(valor)
        else:
            raise ValueError("Estoque não pode ser negativo")
    
    def vender(self, quantidade):
        if quantidade > 0 and quantidade <= self.__estoque:
            self.__estoque -= quantidade
            return f"Venda de {quantidade} unidade(s) realizada."
        return "Estoque insuficiente."

p = Produto("Notebook", 2500, 10)
print(f"Preço: R$ {p.preco:.2f}, Estoque: {p.estoque}")  # Sintaxe elegante
p.vender(3)
print(f"Estoque após venda: {p.estoque}")

3) Classe Pessoa com Idade (com @property)

  1. Crie a classe Pessoa com __idade (private).
  2. Implemente @property e @idade.setter com validação (idade >= 0).
  3. Adicione método fazer_aniversario() que incrementa a idade.
class Pessoa:
    def __init__(self, nome, idade):
        self.nome = nome
        self.__idade = 0
        self.idade = idade  # Usa o setter
    
    @property
    def idade(self):
        return self.__idade
    
    @idade.setter
    def idade(self, valor):
        if valor >= 0:
            self.__idade = int(valor)
        else:
            raise ValueError("Idade não pode ser negativa")
    
    def fazer_aniversario(self):
        self.idade += 1  # Usa o setter automaticamente
        return f"{self.nome} fez aniversário! Agora tem {self.idade} anos."

p = Pessoa("Ana", 25)
print(f"Idade: {p.idade} anos")  # Sintaxe elegante
print(p.fazer_aniversario())
