BEP-019: Encapsulamento e Métodos de Instância

Protegendo dados e criando métodos para manipulação de objetos

Getters e Setters

Métodos para acessar e modificar atributos privados

Getter

Método para ler um atributo privado:

class Pessoa:
    def __init__(self, nome):
        self.__nome = nome
    
    def get_nome(self):  # Getter
        return self.__nome

p = Pessoa("Ana")
print(p.get_nome())  # ✅ Acesso controlado

Padrão: get_atributo()

Setter

Método para modificar um atributo privado:

class Pessoa:
    def __init__(self, nome):
        self.__nome = nome
    
    def set_nome(self, novo_nome):  # Setter
        if isinstance(novo_nome, str) and len(novo_nome) > 0:
            self.__nome = novo_nome
        else:
            raise ValueError("Nome inválido!")

p = Pessoa("Ana")
p.set_nome("Ana Silva")  # ✅ Modificação controlada

Padrão: set_atributo(valor)

Exemplo Completo

class Produto:
    def __init__(self, nome, preco):
        self.__nome = nome
        self.__preco = preco
    
    # Getters
    def get_nome(self):
        return self.__nome
    
    def get_preco(self):
        return self.__preco
    
    # Setters
    def set_nome(self, novo_nome):
        if isinstance(novo_nome, str) and len(novo_nome) > 0:
            self.__nome = novo_nome
        else:
            raise ValueError("Nome inválido!")
    
    def set_preco(self, novo_preco):
        if isinstance(novo_preco, (int, float)) and novo_preco > 0:
            self.__preco = novo_preco
        else:
            raise ValueError("Preço inválido!")

p = Produto("Notebook", 2500)
print(p.get_nome(), p.get_preco())
p.set_preco(2200)
print(p.get_preco())
