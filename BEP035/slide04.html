BEP-035: Introdução ao Django

Compreender o framework Django e sua arquitetura

Criando o primeiro app

Apps organizam funcionalidades dentro do projeto

Criar o app

# criar app
python manage.py startapp core

Ajustes no settings.py

Pontos importantes para o projeto.

# meu_projeto/settings.py

# idioma padrao
LANGUAGE_CODE = "pt-br"

# fuso horario do Brasil
TIME_ZONE = "America/Sao_Paulo"
USE_TZ = True

# banco de dados (padrao: SQLite)
DATABASES = {
    "default": {
        "ENGINE": "django.db.backends.sqlite3",
        "NAME": BASE_DIR / "db.sqlite3",
    }
}

Registrar em settings.py

# settings.py
INSTALLED_APPS = [
    # apps padrao do Django
    "django.contrib.admin",
    "django.contrib.auth",
    "django.contrib.contenttypes",
    "django.contrib.sessions",
    "django.contrib.messages",
    "django.contrib.staticfiles",
    # meu app
    "core",
]

Roteamento (fluxo real)

# meu_projeto/urls.py
from django.contrib import admin
from django.urls import path, include

urlpatterns = [
    path("admin/", admin.site.urls),
    path("", include("core.urls")),
]
# core/urls.py
from django.urls import path
from .views import home

urlpatterns = [
    path("", home),
]
# core/views.py
from django.http import HttpResponse

def home(request):
    return HttpResponse("Ola Django!")
